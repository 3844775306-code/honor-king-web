350 # ==================== 主决策（每0.5秒） ====================
351 
352 func decide() -> void:
353 	try_switch_lane()
354 
355 	match current_phase:
356 		JunglePhase.FARM_HALF, JunglePhase.CLASH_HALF:
357 			# 尝试进入打野模式
358 			if not jungle_mode:
359 				if not _try_enter_jungle_mode():
360 					# 还没到野区 → 朝野区中心移动
361 					hero.move_target = _get_jungle_center()
362 					hero.manual_target = null
363 					return
364 			# 检查野区是否已清空
365 			if _check_jungle_clear():
366 				_advance_phase()
367 				return
368 			# 打野模式：追击目标
369 			var target := _find_jungle_target()
370 			if target:
371 				hero.manual_target = target
372 			else:
373 				_advance_phase()
374 
375 			
376 
377 
378 		JunglePhase.LANE:
379 			_lane_timer += 0.5
380 			# 最多30秒后返回打野
381 			if _lane_timer >= 30.0:
382 				_advance_phase()
383 				return
384 
385 			# 与分路AI相同：多层优先级搜索目标
386 			var lane_target := _find_lane_target()
387 			if lane_target:
388 				hero.manual_target = lane_target
389 				return
390 
391 			# 默认：跟随前方小兵，否则去草丛
392 			var fwd := _get_lane_forward_minion_pos()
393 			if fwd != Vector3.ZERO:
394 				hero.move_target = fwd
395 				hero.manual_target = null
396 			elif _lane_default_bush != Vector3.ZERO:
397 				hero.move_target = _lane_default_bush
398 				hero.manual_target = null
