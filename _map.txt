817 static func _paint_layout() -> Image:
818 	var size := LAYOUT_SIZE
819 	var img := Image.create_empty(size, size, false, Image.FORMAT_RGB8)
820 	var grass := FastNoiseLite.new()
821 	grass.seed = 7
822 	grass.noise_type = FastNoiseLite.TYPE_SIMPLEX_SMOOTH
823 	grass.frequency = 0.045
824 	var patch := FastNoiseLite.new()
825 	patch.seed = 31
826 	patch.noise_type = FastNoiseLite.TYPE_SIMPLEX_SMOOTH
827 	patch.frequency = 0.016
828 	for py in size:
829 		for px in size:
830 			var x: float = (float(px) + 0.5) / float(size) * MAP_HALF * 2.0 - MAP_HALF
831 			var z: float = (float(py) + 0.5) / float(size) * MAP_HALF * 2.0 - MAP_HALF
832 			var n: float = grass.get_noise_2d(x, z) * 0.5 + 0.5          # 0..1 细噪声
833 			var pt: float = patch.get_noise_2d(x, z) * 0.5 + 0.5         # 0..1 大斑块
834 
835 			# 1) 野区草地基底（深浅变化 + 大块阴影）
836 			var col := Color(0.17, 0.31, 0.14).lerp(Color(0.33, 0.48, 0.21), n)
837 			col = col.lerp(Color(0.11, 0.20, 0.11), pt * 0.45)
838 
839 			# 2) 河道河床（沿 x-z=0 对角线与中路垂直；比草地暗、偏青灰）
840 			var river_d: float = absf(x - z) * 0.70710678
841 			col = col.lerp(Color(0.22, 0.28, 0.29).lerp(Color(0.30, 0.36, 0.34), n), _inside(river_d, 13.0, 3.0))
842 
843 			# 3) 三路（土黄石路，双色噪声 + 路缘更深）
844 			var lane: float = _lane_mask(x, z)
845 			if lane > 0.001:
846 				var dirt := Color(0.44, 0.40, 0.30).lerp(Color(0.56, 0.51, 0.38), n)
847 				col = col.lerp(dirt, lane)
848 				var rim: float = _lane_rim(x, z)
849 				if rim > 0.001:
850 					col = col.lerp(Color(0.25, 0.24, 0.19), rim * 0.55)
851 
852 			# 4) 高地 / 基地台座（石砖，最后绘制盖住河道与路）
853 			var hg: float = _high_ground_mask(x, z)
854 			if hg > 0.001:
855 				var stone := Color(0.40, 0.42, 0.36).lerp(Color(0.55, 0.56, 0.48), n)
856 				col = col.lerp(stone, hg * 0.92)
857 
858 			# 5) 地图边缘变暗（远景承接雾色，避免看到"断崖"）
859 			var edge_d: float = maxf(absf(x), absf(z))
860 			col = col.darkened(clampf((edge_d - 48.0) / 14.0, 0.0, 0.4))
861 			img.set_pixel(px, py, col)
862 	return img

864 # 距离场软边：d 在 half±feather 之间平滑过渡
865 static func _inside(d: float, half: float, feather: float) -> float:
866 	return 1.0 - smoothstep(half - feather, half + feather, d)
867 
868 # 三路覆盖度（0..1，软边）：四边路 + 中路对角线（x+z=0 方向，与 _is_on_lane 一致）
869 static func _lane_mask(x: float, z: float) -> float:
870 	var m := 0.0
871 	# 边路贴图一直铺到外墙（±63），与加长后的分路平面一致
872 	var along_z: float = _inside(absf(z), 61.0, 2.0)
873 	var along_x: float = _inside(absf(x), 61.0, 2.0)
874 	m = maxf(m, _inside(absf(absf(x) - 55.0), 8.0, 3.0) * along_z)
875 	m = maxf(m, _inside(absf(absf(z) - 55.0), 8.0, 3.0) * along_x)
876 	m = maxf(m, _inside(absf(x + z) * 0.70710678, 7.5, 2.5))
877 	return m
878 
879 # 路缘（路两侧 1.5 单位内的暗边）
880 static func _lane_rim(x: float, z: float) -> float:
881 	var m := 0.0
882 	var along_z: float = _inside(absf(z), 61.0, 2.0)
883 	var along_x: float = _inside(absf(x), 61.0, 2.0)
884 	m = maxf(m, _inside(absf(absf(absf(x) - 55.0) - 9.5), 1.6, 1.2) * along_z)
885 	m = maxf(m, _inside(absf(absf(absf(z) - 55.0) - 9.5), 1.6, 1.2) * along_x)
886 	m = maxf(m, _inside(absf(absf(x + z) * 0.70710678 - 9.0), 1.6, 1.2))
887 	return m

889 # 高地/基地石材区域：以 (-60,60) 与 (60,-60) 为圆心、半径 35 的扇形范围
890 static func _high_ground_mask(x: float, z: float) -> float:
891 	var d1: float = Vector2(x + 60.0, z - 60.0).length()
892 	var d2: float = Vector2(x - 60.0, z + 60.0).length()
893 	var m: float = maxf(_inside(d1, 34.0, 4.0), _inside(d2, 34.0, 4.0))
894 	return m

953 # 是否是可摆放装饰的开阔地（野区，不在路面/河道/高地/外围）
954 static func _open_ground(x: float, z: float, extra: float = 0.0) -> bool:
955 	if absf(x) > 57.0 or absf(z) > 57.0:
956 		return false
957 	if _lane_mask(x, z) > 0.05:
958 		return false
959 	if absf(x - z) * 0.70710678 < 12.5 + extra:   # 河道（x-z=0 对角线两侧）
960 		return false
