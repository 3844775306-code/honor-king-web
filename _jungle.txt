6 enum JunglePhase { FARM_HALF,  CLASH_HALF, LANE }

8 var hero: Hero

13 var assigned_lane: int = -1             # LANE阶段被分配的具体边路
14 var _phase_cycle: Array = []       # 阶段循环序列
15 var _phase_index: int = 0            # 当前阶段在循环中的索引

21 func _init(p_hero: Hero, p_perception: AiPerception) -> void:
22 	hero = p_hero
23 	perception = p_perception
24 	assigned_lane = AiConfig.Lane.FARM
25 	# 随机选择路线：FARM→LANE→CLASH→LANE 或 CLASH→LANE→FARM→LANE
26 	if randf() < 0.5:
27 		_phase_cycle = [JunglePhase.FARM_HALF, JunglePhase.LANE, JunglePhase.CLASH_HALF, JunglePhase.LANE]
28 	else:
29 		_phase_cycle = [JunglePhase.CLASH_HALF, JunglePhase.LANE, JunglePhase.FARM_HALF, JunglePhase.LANE]
30 	current_phase = _phase_cycle[0]
31 	_phase_index = 0

40 # 判断位置属于下半片还是上半片野区
41 # 下半片：距己方水晶 < 距敌方水晶 且 abs(X) < abs(Z)
42 func _is_bottom_half(pos: Vector3) -> bool:
43 	var own_crystal := Vector3(-53, 0, 53) if hero.team == Config.Team.PLAYER else Vector3(53, 0, -53)
44 	var enemy_crystal := Vector3(53, 0, -53) if hero.team == Config.Team.PLAYER else Vector3(-53, 0, 53)
45 	var closer_to_own := pos.distance_to(own_crystal) < pos.distance_to(enemy_crystal)
46 	var x_less_than_z := pos.x + pos.z >0
47 	return closer_to_own and x_less_than_z

50 # 野区是否已清空（范围内无存活野怪）
51 func _is_jungle_clear(center: Vector3, radius: float) -> bool:
52 	for entity in hero.get_tree().get_nodes_in_group("jungle_monsters"):
53 		if not (entity is JungleMonster) or entity.health <= 0:
54 			continue
55 		if center.distance_to(entity.global_position) <= radius:
56 			return false
57 	return true

60 # 获取当前阶段野区中心点
61 func _get_jungle_center() -> Vector3:
62 	var is_blue := hero.team == Config.Team.PLAYER
63 	match current_phase:
64 		JunglePhase.FARM_HALF:
65 			return Vector3(-20, 0, 40) if is_blue else Vector3(20, 0, -40)
66 		JunglePhase.CLASH_HALF:
67 			return Vector3(-40, 0, 20) if is_blue else Vector3(40, 0, -20)
68 	return Vector3.ZERO

73 # 打野模式目标：附近10m内敌方英雄 > 野区内野怪
74 func _find_jungle_target() -> GameEntity:
75 	# 1. 10m内敌方英雄
76 	if hero.last_attacker and hero.last_attacker is Hero and hero.last_attacker.health >0:
77 		var entity = hero.last_attacker
78 		if not(_is_under_own_tower(entity) and entity.health > 0.2 * entity.max_health) and not entity.is_untargetable_by_enemies():
79 			var dist := hero.global_position.distance_to(entity.global_position)
80 			if dist < 12:
81 				return hero.last_attacker
82 	var best_hero: Hero = null
83 	var best_hp := 99999.0
84 	for entity in hero.get_tree().get_nodes_in_group("heroes"):
85 		if not (entity is Hero) or entity.health <= 0 or entity.team == hero.team:
86 			continue
87 		var dist := hero.global_position.distance_to(entity.global_position)
88 		if dist > AiConfig.JUNGLE_HERO_SEARCH_RANGE:
89 			continue
90 		var h := entity as Hero
91 		if h.is_untargetable_by_enemies():
92 			continue
93 		if h.health < best_hp:
94 			best_hp = h.health
95 			best_hero = h
96 	if best_hero:
97 		return best_hero
98 
99 	# 2. 野区内野怪
100 	var jg_center := _get_jungle_center()
101 	var best_jg: JungleMonster = null
102 	var best_dist := 9999.0
103 	for entity in hero.get_tree().get_nodes_in_group("jungle_monsters"):
104 		if not (entity is JungleMonster) or entity.health <= 0:
105 			continue
106 		var m := entity as JungleMonster
107 		if not _is_bottom_half(m.global_position) and current_phase == JunglePhase.FARM_HALF:
108 			continue
109 		if _is_bottom_half(m.global_position) and current_phase == JunglePhase.CLASH_HALF:
110 			continue
111 		var dist := hero.global_position.distance_to(m.global_position)
112 		if dist < best_dist:
113 			best_dist = dist
114 			best_jg = m
115 	return best_jg

118 # 线上模式目标（含塔安全检测）
119 func _find_lane_target() -> GameEntity:
120 	# 1. enemy heroes (not under own tower, not stealthed/invincible)
121 	var best_hero: Hero = null
122 	var best_hero_hp := 99999.0
123 	for entity in hero.get_tree().get_nodes_in_group("heroes"):
124 		if not (entity is Hero) or entity.health <= 0 or entity.team == hero.team:
125 			continue
126 		var dist := hero.global_position.distance_to(entity.global_position)
127 		if dist > AiConfig.LANE_SEARCH_RANGE:
128 			continue
129 		var h := entity as Hero
130 		if h.is_untargetable_by_enemies():
131 			continue
132 		if _is_under_own_tower(h) and not AiConfig.should_engage_tower_hero(hero, h):
133 			continue
134 		if h.health < best_hero_hp:
135 			best_hero_hp = h.health
136 			best_hero = h
137 	if best_hero:
138 		return best_hero
139 
140 	# 2. enemy minions (not under unsafe enemy tower)
141 	var best_minion: Minion = null
142 	var best_minion_hp := 99999.0
143 	for entity in hero.get_tree().get_nodes_in_group("minions"):
144 		if not (entity is Minion) or entity.health <= 0 or entity.team == hero.team:
145 			continue
146 		var dist := hero.global_position.distance_to(entity.global_position)
147 		if dist > AiConfig.LANE_SEARCH_RANGE:
148 			continue
149 		if _is_under_unsafe_tower(entity):
150 			continue
151 		var m := entity as Minion
152 		if m.health < best_minion_hp:
153 			best_minion_hp = m.health
154 			best_minion = m
155 	if best_minion:
156 		return best_minion
157 
158 	# 3. enemy turrets (after 4min + minion cover)
159 	if game_time >= AiConfig.EARLY_GAME_END:
160 		var best_tower: Turret = null
161 		var best_tower_dist := 9999.0
162 		for entity in hero.get_tree().get_nodes_in_group("turrets"):
163 			if not (entity is Turret) or entity.health <= 0 or entity.team == hero.team:
164 				continue
165 			var dist := hero.global_position.distance_to(entity.global_position)
166 			if dist > AiConfig.LANE_SEARCH_RANGE:
167 				continue
168 			var friendly_hp := 0.0
169 			for m in hero.get_tree().get_nodes_in_group("minions"):
170 				if not (m is Minion) or m.health <= 0 or m.team != hero.team:
171 					continue
172 				var td = entity.global_position.distance_to(m.global_position)
173 				if td <= AiConfig.TOWER_ATTACK_RANGE:
174 					friendly_hp += (m as Minion).health
175 			if friendly_hp < AiConfig.UNSAFE_TOWER_MINION_HP:
176 				continue
177 			if dist < best_tower_dist:
178 				best_tower_dist = dist
179 				best_tower = entity as Turret
180 		if best_tower:
181 			return best_tower
182 
183 	return null
184 # 实体是否处于敌方塔射程内且无敌方小兵掩护
185 func _is_under_unsafe_tower(entity: GameEntity) -> bool:
186 	for t in hero.get_tree().get_nodes_in_group("turrets"):
187 		if not (t is Turret) or t.health <= 0 or t.team == hero.team:
188 			continue
189 		var turret := t as Turret
190 		if entity.global_position.distance_to(turret.global_position) > AiConfig.TOWER_ATTACK_RANGE:
191 			continue
192 		var friendly_hp := 0.0
193 		for m in hero.get_tree().get_nodes_in_group("minions"):
194 			if not (m is Minion) or m.health <= 0 or m.team != hero.team:
195 				continue
196 			if turret.global_position.distance_to(m.global_position) <= AiConfig.TOWER_ATTACK_RANGE:
197 				friendly_hp += (m as Minion).health
198 		if friendly_hp < AiConfig.UNSAFE_TOWER_MINION_HP:
199 			return not AiConfig.can_hit_from_outside_tower(hero, entity.global_position, float(entity.body_radius))
200 	return false

202 # 实体是否处于己方（该实体阵营的）塔下
203 func _is_under_own_tower(entity: GameEntity) -> bool:
204 	for t in hero.get_tree().get_nodes_in_group("turrets"):
205 		if not (t is Turret) or t.health <= 0 or t.team != entity.team:
206 			continue
207 		if entity.global_position.distance_to(t.global_position) <= AiConfig.TOWER_ATTACK_RANGE:
208 			return true
209 	return false
210 func _try_enter_jungle_mode() -> bool:
211 	if jungle_mode:
212 		return true
213 	# 检查是否到达当前阶段的野区
214 	var jg_center := _get_jungle_center()
215 	var dist := hero.global_position.distance_to(jg_center)
216 	if dist <= 30.0:  # 接近野区
217 		if not _is_jungle_clear(jg_center, 50.0):
218 			jungle_mode = true
219 			return true
220 	return false
221 
222 
223 func _check_jungle_clear() -> bool:
224 	if not jungle_mode:
225 		return true
226 	var jg_center := _get_jungle_center()
227 	if _is_jungle_clear(jg_center, 50.0):
228 		jungle_mode = false
229 		_advance_phase()
230 		return true
231 	return false
232

235 func _advance_phase() -> void:
236 	_phase_index = (_phase_index + 1) % _phase_cycle.size()
237 	current_phase = _phase_cycle[_phase_index]
238 	jungle_mode = false
239 	_lane_timer = 0.0
240 	if current_phase == JunglePhase.LANE:
241 		assigned_lane = _get_nearest_enemy_lane()
242 		_lane_default_bush = _get_lane_default_bush()

246 # 选择敌方小兵离自己最近的分路
247 func _get_nearest_enemy_lane() -> int:
248 	var lanes := [AiConfig.Lane.FARM, AiConfig.Lane.CLASH,AiConfig.Lane.MID]
249 	var best_lane := AiConfig.Lane.FARM
250 	var best_dist := 99999.0
251 	for ln in lanes:
252 		for entity in hero.get_tree().get_nodes_in_group("minions"):
253 			if not (entity is Minion) or entity.health <= 0 or entity.team == hero.team:
254 				continue
255 			var m := entity as Minion
256 			if m.line != ln:
257 				continue
258 			var dist := hero.global_position.distance_to(m.global_position)
259 			if dist < best_dist:
260 				best_dist = dist
